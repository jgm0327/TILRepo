            import java.io.BufferedReader;
            import java.io.IOException;
            import java.io.InputStreamReader;
            import java.util.StringTokenizer;

            class Solution {
                private static int[] weights;
                private static boolean[] visit;
                private static int count, n;
                
                public static void main(String args[]) throws IOException {
                    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

                    int T = Integer.parseInt(br.readLine());

                    StringBuilder answer = new StringBuilder();
                    for(int test_case = 1 ; test_case <= T ; test_case++){
                        n = Integer.parseInt(br.readLine());

                        StringTokenizer tokenizer = new StringTokenizer(br.readLine());
                        weights = new int[n];
                        for(int i=0 ; i<n ; i++){
                            weights[i] = Integer.parseInt(tokenizer.nextToken());
                        }

                        count = 0;
                        visit = new boolean[n];
                        dfs(n, 0, 0);

                        answer.append("#").append(test_case).append(" ").append(count).append("\n");
                    }

                    System.out.println(answer);
                    br.close();
                }

                private static void dfs(int depth, int left, int right){
                    if(0 == depth){
                        count++;
                        return;
                    }

                    for(int i=0 ; i<n ; i++){
                        if(visit[i])
                            continue;

                        visit[i] = true;
                        if(right + weights[i] <= left)
                            dfs(depth - 1, left, right + weights[i]);
                        dfs(depth - 1, left + weights[i], right);

                        visit[i] = false;
                    }
                }

            }
