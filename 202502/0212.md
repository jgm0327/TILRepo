            class Solution {
                public int maximumSum(int[] nums) {
                    PriorityQueue<int[]> pq = new PriorityQueue<>((o1, o2) -> {
                        if(o1[0] != o2[0])return o1[0] - o2[0];
                        return o2[1] - o1[1];
                    });

                    for(int num : nums){
                        pq.add(new int[]{sumOfDigits(num), num});
                    }

                    int answer = -1;
                    while(!pq.isEmpty()){
                        int[] a = pq.poll();
                        int digits = a[0];

                        if(pq.isEmpty())
                            break;

                        if(digits != pq.peek()[0])
                            continue;

                        int[] b = pq.poll();
                        answer = Math.max(answer, a[1] + b[1]);

                        while(!pq.isEmpty() && pq.peek()[0] == digits){
                            pq.poll();
                        }
                    }
                    
                    return answer;
                }

                private static int sumOfDigits(int number){
                    int sum = 0;
                    while(number > 0){
                        sum += number % 10;
                        number /= 10;
                    }
                    return sum;
                }
            }

            class Solution {
                public int maximumSum(int[] nums) {
                    int[] maxValue = new int[82];
                    int answer = -1;
                    for(int num : nums){
                        int digits = sumOfDigits(num);

                        if(maxValue[digits] != 0){
                            answer = Math.max(answer, maxValue[digits] + num);
                        }

                        maxValue[digits] = Math.max(maxValue[digits], num);
                    }

                    return answer;
                }

                private static int sumOfDigits(int number){
                    int sum = 0;
                    while(number > 0){
                        sum += number % 10;
                        number /= 10;
                    }
                    return sum;
                }
            }
