            import java.io.BufferedReader;
            import java.io.IOException;
            import java.io.InputStreamReader;

            class Solution {
                private static int n, l, maxScore;
                private static int[][] hamburgers;

                public static void main(String args[]) throws IOException {
                    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

                    int T = Integer.parseInt(br.readLine());

                    StringBuilder answer = new StringBuilder();
                    for (int test_case = 1; test_case <= T; test_case++) {
                        String[] split = br.readLine().split(" ");

                        n = Integer.parseInt(split[0]);
                        l = Integer.parseInt(split[1]);

                        hamburgers = new int[n][2];
                        for (int i = 0; i < n; i++) {
                            split = br.readLine().split(" ");
                            int taste = Integer.parseInt(split[0]);
                            int cal = Integer.parseInt(split[1]);
                            hamburgers[i] = new int[] { taste, cal };
                        }

                        maxScore = 0;
                        dfs(0, 0, 0, new boolean[n]);
                        answer.append("#").append(test_case).append(" ").append(maxScore).append("\n");
                    }

                    System.out.print(answer);
                    br.close();
                }

                private static void dfs(int totalCal, int totalScore, int start, boolean[] visit) {
                    for (int i = start; i < n; i++) {
                        if (visit[i] || totalCal + hamburgers[i][1] > l)
                            continue;

                        visit[i] = true;
                        maxScore = Math.max(maxScore, totalScore + hamburgers[i][0]);
                        dfs(totalCal + hamburgers[i][1], totalScore + hamburgers[i][0], i + 1, visit);
                        visit[i] = false;
                    }
                }

            }
