            import java.io.BufferedReader;
            import java.io.IOException;
            import java.io.InputStreamReader;
            import java.util.ArrayDeque;
            import java.util.Queue;
            import java.util.StringTokenizer;

            class Solution {
                static int n;
                static int[] dy = { 0, 0, 1, -1 }, dx = { 1, -1, 0, 0 };
                static int[][] rooms;

                public static void main(String args[]) throws IOException {
                    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

                    int T = Integer.parseInt(br.readLine());
                    StringBuilder answer = new StringBuilder();
                    for (int test_case = 1; test_case <= T; test_case++) {
                        n = Integer.parseInt(br.readLine());

                        rooms = new int[n][n];
                        for (int i = 0; i < n; i++) {
                            StringTokenizer tokenizer = new StringTokenizer(br.readLine());
                            for (int j = 0; j < n; j++) {
                                rooms[i][j] = Integer.parseInt(tokenizer.nextToken());
                            }
                        }

                        int maxValue = 0, roomNumber = 1;
                        for (int i = 0; i < n; i++) {
                            for (int j = 0; j < n; j++) {
                                int value = bfs(i, j);

                                if (value > maxValue) {
                                    maxValue = value;
                                    roomNumber = rooms[i][j];
                                }
                                else if(value == maxValue)
                                    roomNumber = Math.min(roomNumber, rooms[i][j]);
                            }
                        }

                        answer.append("#").append(test_case).append(" ")
                                .append(roomNumber).append(" ").append(maxValue).append("\n");
                    }

                    System.out.println(answer);
                    br.close();
                }

                static int bfs(int startY, int startX) {
                    int total = 0;
                    Queue<int[]> que = new ArrayDeque<>();
                    que.add(new int[] { startY, startX });

                    while (!que.isEmpty()) {
                        int[] cur = que.poll();

                        int y = cur[0], x = cur[1];

                        total++;
                        for (int i = 0; i < 4; i++) {
                            int ny = y + dy[i], nx = x + dx[i];

                            if (0 > ny || ny >= n || 0 > nx || nx >= n
                                    || rooms[ny][nx] != rooms[y][x] + 1)
                                continue;

                            que.add(new int[] { ny, nx });
                        }
                    }

                    return total;
                }
            }
